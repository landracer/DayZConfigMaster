500 Object actionTargetObject = null;

534 {
535 if ( !ctx.Read(proxyBoneIdx) )
536 return false;
537
538 if ( !ctx.Read(actionTargetParent))
539 return false;
540
541 if ( !ctx.Read(componentIndex) )
542 return false;
543
544 if ( !ctx.Read(cursorHitPos) )
545 return false;
546
548 if ( proxyBoneIdx > -1 )
