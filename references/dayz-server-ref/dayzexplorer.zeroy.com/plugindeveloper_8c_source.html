896 int QuickSortPartition( TStringArray arr, int left, int right )
897 {
898 string pivot = arr.Get( left );
899 int i = left;
900 int j = right + 1;
901 string temp;
902
903 while ( true )
904 {
905 while ( true )
906 {
907 i++;
908 if ( i > right || arr.Get(i) > pivot )
909 {
910 break;
911 }
912 }
913
914 while ( true )
915 {
916 j--;
917 if ( arr.Get(j) <= pivot )
918 {
919 break;
920 }
921 }
922
923 if ( i >= j )
924 {
925 break;
926 }
927
928 temp = arr.Get( i );
929 arr.Set( i, arr.Get(j) );
930 arr.Set( j, temp );
931 }
932
933 temp = arr.Get( left );
934 arr.Set( left, arr.Get(j) );
935 arr.Set( j, temp );
936
937 return j;
938 }

947 QuickSort( arr, left, j - 1 );
