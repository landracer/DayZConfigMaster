379 void DumpWidget(Widget w, int tabs)
380 {
381 if ( !w )
382 {
383 return;
384 }
385
386 m_WidgetsTotal++;
387
388 string tmp;
389 for (int i = 0; i < tabs; i++)
390 {
391 tmp += " ";
392 }
393
394 string invisible = "";
395
396 if ( !w.IsVisibleHierarchy() )
397 {
398 invisible = "[invisible]";
399 m_WidgetsInvisible++;
400 }
401

403
